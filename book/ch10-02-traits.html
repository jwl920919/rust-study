}

pub struct NewsArticle {
    pub headline: String,
    pub location: String,
    pub author: String,
    pub content: String,
}

impl Summary for NewsArticle {
    fn summarize(&self) -> String {
        format!("{}, by {} ({})", self.headline, self.author, self.location)

pub struct SocialPost {

pub fn notify(item: &impl Summary) {

Trait 바인드 구문

impl Trait 구문은 간단한 경우에 작동하지만 실제로는 특성 경계로 알려진 더 긴 형태의 구문 설탕입니다.

pub fn notify<T: Summary>(item: &T) {
    println!("Breaking news! {}", item.summarize());
}

이 긴 형식은 이전 섹션의 예제와 동일하지만 더 verbose입니다. 우리는 콜론과 내부 각도 브래킷 뒤에 일반 유형 매개 변수의 선언으로 특성 경계를 배치합니다.

impl Trait 구문은 편리하고 간단한 경우보다 간결한 코드를 만드는 반면, 완전한 트레잇 바인드 구문은 다른 경우보다 복잡성을 표현할 수 있습니다. 예를 들어 Summary를 구현하는 두 가지 매개 변수를 가질 수 있습니다. impl Trait 구문은 다음과 같습니다.

pub fn notify(item1: &impl Summary, item2: &impl Summary) {

impl Trait를 사용하는 것은 이 함수가 item1 및 item2가 서로 다른 유형을 갖도록 허용하려면 적합합니다(두 유형 모두 Summary를 구현하는 한). 그러나 두 매개 변수 모두 동일한 유형을 갖도록 강제하려면 다음과 같이 바인딩된 특성을 사용해야 합니다.

pub fn notify<T: Summary>(item1: &T, item2: &T) {

제네릭 타입 T 상기 타입은 The type of the item1 그리고 item2 매개 변수는 함수를 제한하여 특정 유형의 값이 인수로 전달되도록 합니다. item1 그리고 item2 같은 것이어야 한다.

하나 이상의 트레잇 바운드를 지정할 수도 있습니다. notify 표시 형식뿐만 아니라 사용 summarize 온 item: 우리는에 지정 notify definition 그것은 item 둘 다 실행해야 합니다. Display 그리고 Summary. 우리는을 사용하여 그렇게 할 수 있습니다 + 문법:

더 명확한 Trait 경계 where 절

너무 많은 트레잇 바운드를 사용하는 것은 단점을 가지고 있습니다. 각 제네릭은 자체 트레잇 바운드를 가지고 있으므로, 여러 제네릭 타입 파라미터를 가진 함수는 함수의 이름과 파라미터 목록 사이에 트레잇 바인드 정보를 많이 포함할 수 있어, 함수의 서명을 읽기 어렵게 만듭니다. 이런 이유로, 러스트는 함수 서명 뒤에 where 절 안에 트레잇 바운드를 지정하기 위한 대체 문법을 가지고 있습니다. 따라서, 이것을 쓰는 대신:

fn some_function<T: Display + Clone, U: Clone + Debug>(t: &T, u: &U) -> i32 {

다음과 같이 where 절을 사용할 수 있습니다.

fn some_function<T, U>(t: &T, u: &U) -> i32
where
    T: Display + Clone,
    U: Clone + Debug,
{
    unimplemented!()
}

이 함수의 서명은 덜 복잡합니다: 함수 이름, 매개 변수 목록 및 반환 형식은 많은 특성 경계가 없는 함수와 유사하게 서로 가깝습니다.

특성을 구현하는 반환 형식Returning Types that Implement Traits

또한 반환 위치에서 impl Trait 구문을 사용하여 트레잇을 구현하는 일부 유형의 값을 반환할 수도 있습니다.

pub trait Summary {
    fn summarize(&self) -> String;
}

pub struct NewsArticle {
    pub headline: String,
    pub location: String,
    pub author: String,
    pub content: String,
}

impl Summary for NewsArticle {
    fn summarize(&self) -> String {
        format!("{}, by {} ({})", self.headline, self.author, self.location)
    }
}

pub struct SocialPost {
    pub username: String,
    pub content: String,
    pub reply: bool,
    pub repost: bool,
}

impl Summary for SocialPost {
    fn summarize(&self) -> String {
        format!("{}: {}", self.username, self.content)
    }
}

fn returns_summarizable() -> impl Summary {
    SocialPost {
        username: String::from("horse_ebooks"),
        content: String::from(
            "of course, as you probably already know, people",
        ),
        reply: false,
        repost: false,
    }
}

반환 형식에 impl Summary를 사용하여 returns_summarizable 함수가 구체적인 형식 이름을 지정하지 않고 Summary 특성을 구현하는 일부 형식을 반환하도록 지정합니다. 이 경우 returns_summarizable는 SocialPost를 반환하지만 이 함수를 호출하는 코드는 이를 알 필요가 없습니다.

구현한 트레잇에 의해서만 리턴 타입을 지정하는 기능은 특히 클로저와 이터레이터의 컨텍스트에서 유용합니다. 이터레이터는 13장에서 다룹니다. 클로저와 이터레이터는 컴파일러만 알고 있는 타입이나 매우 긴 타입만을 만듭니다. impl Trait 구문을 사용하면 함수가 매우 긴 타입을 작성할 필요 없이 Iterator 트레잇을 구현하는 어떤 타입을 반환하도록 간결하게 지정할 수 있습니다.

그러나 단일 유형을 반환하는 경우에만 impl Trait를 사용할 수 있습니다. 예를 들어 NewsArticle 또는 SocialPost를 반환하는이 코드는 impl Summary로 지정된 반환 유형이 작동하지 않습니다.

pub struct NewsArticle {

impl Summary for NewsArticle {

pub struct SocialPost {

fn returns_summarizable(switch: bool) -> impl Summary {

NewsArticle 또는 SocialPost를 반환하는 것은 impl Trait 구문이 컴파일러에서 구현되는 방법에 대한 제한으로 인해 허용되지 않습니다. 18 장의 "공유 된 행동을 추상화하기 위해 Trait 객체 사용" 섹션에서이 동작으로 함수를 작성하는 방법을 설명합니다.

Trait Bounds를 사용하여 조건부 구현 방법

상기 결합하는 단계는, Wherein a bound with the impl 제네릭 타입 파라미터를 사용하는 블록은 지정된 트레잇을 구현하는 타입에 대해 조건부로 메소드를 구현할 수 있습니다. 예를 들어, 타입 Pair<T> Listing 10-15에서는 항상 new 새로운 인스턴스를 반환하는 함수 Pair<T> 상기 수신하는 단계는, Recalling from the “방법 의 통념” 무시 CHAPTER 5 그것은 Self 상기 타입에 대한 별칭은 The alias for the type of the impl 블록은, In this case, Pair<T>)). 하지만 다음에 impl 블록, Pair<T> 단지 그것을 실행한다. cmp_display 상기 내부 타입은 Wherein the inner type T 를 구현한다. PartialOrd 비교를 가능하게 하는 특성 그리고 오시는 길 Display 인쇄를 가능하게 하는 제품.

우리는 또한 다른 트레잇을 구현하는 모든 타입에 대한 트레잇을 조건부로 구현할 수 있습니다. 트레잇 범위를 만족하는 모든 타입에 대한 트레잇의 구현은 블랭킷 구현이라고 불리며 러스트 표준 라이브러리에서 광범위하게 사용됩니다. 예를 들어, 표준 라이브러리는 ToString 트레잇을 구현하는 모든 타입에 Display 트레잇을 구현합니다. 표준 라이브러리의 impl 블록은 다음과 유사합니다.

impl<T: Display> ToString for T {
    // --snip--
}

표준 라이브러리는 이 블랭킷 구현을 가지고 있기 때문에 to_string 상기 정의된 방법은 The method defined by the ToString 를 구현하는 모든 유형의 특성Teature on any type that implements the Display 예를 들어, 정수를 해당 정수로 변환할 수 있습니다. String 정수가 구현하기 때문에 이와 같은 값 Display:

#![allow(unused)]
fn main() {
let s = 3.to_string();
}

Blanket 구현은 “Implementors” 섹션의 특성에 대한 문서에 나타납니다.

특성과 트레잇 바운드는 제네릭 타입 파라미터를 사용하여 복제를 줄이기 위해 코드를 작성할 수 있지만 제네릭 타입이 특정 동작을 가지도록 컴파일러에 지정할 수 있습니다. 그러면 컴파일러는 트레잇 바인드 정보를 사용하여 코드와 함께 사용되는 모든 콘크리트 타입이 올바른 동작을 제공하는지 확인할 수 있습니다. 또한, 우리가 메소드를 정의하지 않은 타입에 대한 메소드를 호출하면 런타임에 오류가 발생합니다. 그러나 러스트는 우리가 컴파일하기 전에 이러한 오류를 수정하기 위해 이러한 오류를 이동시킵니다.
